# a rare metal; a legendary one if every rare metal is known here; half the usual gold if none is left
DA_GX_rare_metal_or_better = {
	if = {
		limit = { DA_prov_can_find_metal_rare = yes }
		hidden_effect = { set_variable = { which = DA_else_DA_GX_rare_metal_or_better_1 value = 1 } }
		DA_add_Metal_Rare = yes
	}
	if = {
		limit = { NOT = { check_variable = { which = DA_else_DA_GX_rare_metal_or_better_1 value = 1 } } NOT = { AND = { DA_prov_can_find_metal_rare = yes } } DA_prov_can_find_metal_legendary = yes }
		hidden_effect = { set_variable = { which = DA_else_DA_GX_rare_metal_or_better_1 value = 1 } }
		DA_add_Metal_Legendary = yes
	}
	if = {
		limit = { NOT = { check_variable = { which = DA_else_DA_GX_rare_metal_or_better_1 value = 1 } } NOT = { AND = { DA_prov_can_find_metal_rare = yes } } NOT = { AND = { DA_prov_can_find_metal_legendary = yes } } }
		DA_GX_half_gold = yes
	}
	hidden_effect = { set_variable = { which = DA_else_DA_GX_rare_metal_or_better_1 value = 0 } }
}

# A company is lost: half the time some of them come back in 9 to 15 months with their maps (DA_ADK_explore.40).
# The flag is cleared on conquest (DA_HX_reset), so they come back only to the owner that sent them.
DA_GX_company_lost = {
	custom_tooltip = DA_GX_company_lost_tt
	hidden_effect = {
		random = {
			chance = 50
			set_province_flag = DA_GX_lost_company
			province_event = { id = DA_ADK_explore.40 days = 270 random = 180 }
		}
	}
}

# --- Strâmolgiv: one barrel at a time is brewed (flag DA_ST_brewing, variable DA_ST_brew_months); the cellar is the variable DA_ST_barrels
DA_ST_begin_brew = {
	add_treasury = -150
	custom_tooltip = DA_ST_begin_brew_tt
	hidden_effect = {
		set_country_flag = DA_ST_brewing
		set_variable = { which = DA_ST_brew_months value = 12 }
		if = {
			limit = { DA_ST_has_grain = yes }
			set_variable = { which = DA_ST_brew_months value = 8 }
		}
	}
}
DA_ST_monthly = {
	change_variable = { which = DA_ST_barrels value = 0 }
	if = {
		limit = { has_country_flag = DA_ST_brewing }
		subtract_variable = { which = DA_ST_brew_months value = 1 }
		if = {
			limit = { NOT = { check_variable = { which = DA_ST_brew_months value = 1 } } }
			clr_country_flag = DA_ST_brewing
			change_variable = { which = DA_ST_barrels value = 1 }
		}
	}
	if = { # the ruling on Darkbarrel's kegs: tier III, and the cellar found
		limit = {
			DA_ST_at_tier_3 = yes
			4266 = {
				has_province_flag = DA_Krakazol_exp3
				NOT = { has_province_flag = DA_ST_ruling_asked }
			}
		}
		4266 = { set_province_flag = DA_ST_ruling_asked }
		country_event = { id = DA_Remaining_Wonders_event.105 }
	}
}

# --- the Magistrates' Vault: DA_MV_gold lies in the vault; DA_MV_cap is what it holds, DA_MV_room what is free; the mark is how full it is, in quarters
DA_MV_sync = {
	change_variable = { which = DA_MV_gold value = 0 }
	set_variable = { which = DA_MV_cap value = 0 }
	if = {
		limit = { DA_MV_at_tier_1 = yes }
		set_variable = { which = DA_MV_cap value = 2000 }
	}
	if = {
		limit = { DA_MV_at_tier_2 = yes }
		set_variable = { which = DA_MV_cap value = 4000 }
	}
	if = {
		limit = { DA_MV_at_tier_3 = yes }
		set_variable = { which = DA_MV_cap value = 8000 }
		if = {
			limit = { 2914 = { has_province_flag = DA_MV_off_books } }
			change_variable = { which = DA_MV_cap value = 4000 }
		}
	}
	set_variable = { which = DA_MV_room which = DA_MV_cap }
	subtract_variable = { which = DA_MV_room which = DA_MV_gold }
	set_variable = { which = DA_MV_mark value = 0 }
	if = {
		limit = { check_variable = { which = DA_MV_cap value = 1 } }
		set_variable = { which = DA_MV_mark which = DA_MV_gold }
		multiply_variable = { which = DA_MV_mark value = 4 }
		divide_variable = { which = DA_MV_mark which = DA_MV_cap }
	}
	if = {
		limit = {
			DA_MV_at_tier_1 = yes
			check_variable = { which = DA_MV_mark value = 1 }
			NOT = { check_variable = { which = DA_MV_mark value = 2 } }
			NOT = { has_country_modifier = DA_MV_mark_1 }
		}
		add_country_modifier = {
			name = DA_MV_mark_1
			duration = -1
		}
	}
	if = {
		limit = {
			has_country_modifier = DA_MV_mark_1

				AND = {
					DA_MV_at_tier_1 = yes
					check_variable = { which = DA_MV_mark value = 1 }

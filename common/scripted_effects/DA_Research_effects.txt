

# effects used in periodical research events adding/substracting progress

# Target changes only at research start/reset. Refresh on load for existing saves.
DA_refresh_research_event_cutoff = {
	# Existing saves can have active research without a generation counter yet.
	if = {
		limit = { NOT = { check_variable = { which = DA_research_generation value = 1 } } }
		set_variable = { which = DA_research_generation value = 1 }
	}
	set_variable = { which = DA_research_event_cutoff which = DA_research_progress_MAX }
	subtract_variable = { which = DA_research_event_cutoff value = 5 }
}

DA_capture_research_event = {
	hidden_effect = {
		set_country_flag = DA_research_event_pending.$id$
		set_variable = {
			which = DA_research_event_generation.$id$
			which = DA_research_generation
		}
	}
}

DA_Research_add = {
	custom_tooltip = DA_Research_add.$amount$
	hidden_effect = {
		change_variable = {
			which = DA_research_progress
			value = $amount$
		}
	}
}
DA_Research_lose = {
	custom_tooltip = DA_Research_lose.$amount$
	hidden_effect = {
		subtract_variable = {
			which = DA_research_progress
			value = $amount$
		}
		if = {
			limit = {
				NOT = {
					check_variable = {
						which = DA_research_progress
						value = 0
					}
				}
			}
			set_variable = {
				which = DA_research_progress
				value = 0
			}
		}
	}
}
DA_Research_terminate = {
	custom_tooltip = DA_Research_terminate_tt
	hidden_effect = {
		DA_reset_research = yes
		remove_estate_privilege = DA_Runesmith_research_privilege
	}
}
DA_Research_reset = {
	custom_tooltip = DA_Research_reset_tt
	hidden_effect = {
		set_variable = {
			which = DA_research_progress
			value = 0
		}
	}
}
DA_Research_complete = {
	custom_tooltip = DA_Research_complete_tt
	hidden_effect = {
		set_variable = {
			which = DA_research_progress
			which = DA_research_progress_MAX
		}
	}
}

# Restarts the cooldown of a random research event. Clear first, or the flag keeps its old date.
DA_research_event_seen = {
	hidden_effect = {
		clr_country_flag = DA_research_event_seen.$id$
		set_country_flag = DA_research_event_seen.$id$
	}
}

# school = Alchemy | Arcane_Arts | Mechanics | Medicine | Metallurgy
DA_research_school_exp = {
	custom_tooltip = DA_research_school_exp_tt.$school$
	hidden_effect = {
		change_variable = { which = DA_$school$_exp value = 250 }
	}
}

# Honouring a people of race $race$: a small tolerance gain when it is not the race of our primary culture.
DA_research_race_honour = {
	if = {
		limit = { NOT = { culture_is_$race$ = yes } }
		small_increase_of_$race$_tolerance_effect = yes
	}
}

# The same, or prestige when it is our own race.
DA_research_race_honour_or_prestige = {
	if = {
		limit = { culture_is_$race$ = yes }
		add_prestige = 5
	}
	if = {
		limit = { NOT = { culture_is_$race$ = yes } }
		small_increase_of_$race$_tolerance_effect = yes
	}
}

# Slighting a people of race $race$: prestige when it is our own race, a small tolerance loss otherwise.
DA_research_race_slight = {
	if = {
		limit = { culture_is_$race$ = yes }
		add_prestige = -10
	}
	if = {
		limit = { NOT = { culture_is_$race$ = yes } }
		small_decrease_of_$race$_tolerance_effect = yes
	}
}

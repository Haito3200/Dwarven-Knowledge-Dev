

# effects used in periodical research events adding/substracting progress

# Target changes only at research start/reset. Refresh on load for existing saves.
DA_refresh_research_event_cutoff = {
	# Existing saves can have active research without a generation counter yet.
	if = {
		limit = { NOT = { check_variable = { which = DA_research_generation value = 1 } } }
		set_variable = { which = DA_research_generation value = 1 }
	}
	set_variable = { which = DA_research_event_cutoff which = DA_research_progress_MAX }
	subtract_variable = { which = DA_research_event_cutoff value = 5 }
}

DA_capture_research_event = {
	hidden_effect = {
		set_country_flag = DA_research_event_pending.$id$
		set_variable = {
			which = DA_research_event_generation.$id$
			which = DA_research_generation
		}
	}
}

DA_Research_add = {
	custom_tooltip = DA_Research_add.$amount$
	hidden_effect = {
		change_variable = {
			which = DA_research_progress
			value = $amount$
		}
	}
}
DA_Research_lose = {
	custom_tooltip = DA_Research_lose.$amount$
	hidden_effect = {
		subtract_variable = {
			which = DA_research_progress
			value = $amount$
		}
		if = {
			limit = {
				NOT = {
					check_variable = {
						which = DA_research_progress
						value = 0
					}
				}
			}
			set_variable = {
				which = DA_research_progress
				value = 0
			}
		}
	}
}
DA_Research_terminate = {
	custom_tooltip = DA_Research_terminate_tt
	hidden_effect = {
		DA_reset_research = yes
		remove_estate_privilege = DA_Runesmith_research_privilege
	}
}
DA_Research_reset = {
	custom_tooltip = DA_Research_reset_tt
	hidden_effect = {
		set_variable = {
			which = DA_research_progress
			value = 0
		}
	}
}
DA_Research_complete = {
	custom_tooltip = DA_Research_complete_tt
	hidden_effect = {
		set_variable = {
			which = DA_research_progress
			which = DA_research_progress_MAX
		}
	}
}

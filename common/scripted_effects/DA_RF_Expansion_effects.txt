# button's trigger (and DA_AM_rf_can_expand, built from it); this effect only pays and grants.

# Third fix (Claude files/tools/apply_rf_level_if_2026-11-16.py): the cases are plain ifs, not else_if. EU4 drops a

		}

			type = Hold_restorer

		}
		hidden_effect = {
			set_variable = { which = DA_RF_next_level value = 3 }
			clr_country_flag = DA_RF_level_pending	# a flag left by a lost event starts again from today
			set_country_flag = DA_RF_level_pending	# until DA_Government_event.2 has granted the level: no second payment

# A level was paid for, but its event has not come in 30 days (DA_RF_level_pending is older than that): send
# DA_Government_event.2 again, free. Called by the Expansion button before DA_RF_expand_one_level, which then sees
# the flag as new and does not charge. (Claude files/tools/apply_rf_expansion_event_2026-11-16.py)
DA_RF_resend_level = {
	if = {
		limit = {
			has_country_flag = DA_RF_level_pending
			had_country_flag = { flag = DA_RF_level_pending days = 30 }
			check_variable = { which = DA_RF_next_level value = 3 }
		}
		hidden_effect = {
			clr_country_flag = DA_RF_level_pending
			set_country_flag = DA_RF_level_pending
		}
		country_event = { id = DA_Government_event.2 days = 1 }
	}
}

# Called only from events: dice thrown from a button always fall the same way.
DA_SZ_roll_first_pod = {
	clr_country_flag = DA_SZ_roll_lost
	if = {
		limit = { has_country_flag = DA_SZ_pod_lost_2 }
	}
	else_if = {
		limit = { DA_SZ_first_pod_precautions_5 = yes }
		random = { chance = 4 set_country_flag = DA_SZ_roll_lost }
	}
	else_if = {
		limit = { DA_SZ_first_pod_precautions_4 = yes }
		random = { chance = 8 set_country_flag = DA_SZ_roll_lost }
	}
	else_if = {
		limit = { DA_SZ_first_pod_precautions_3 = yes }
		random = { chance = 16 set_country_flag = DA_SZ_roll_lost }
	}
	else_if = {
		limit = { DA_SZ_first_pod_precautions_2 = yes }
		random = { chance = 24 set_country_flag = DA_SZ_roll_lost }
	}
	else_if = {
		limit = { DA_SZ_first_pod_precautions_1 = yes }
		random = { chance = 32 set_country_flag = DA_SZ_roll_lost }
	}
	else = {
		random = { chance = 40 set_country_flag = DA_SZ_roll_lost }
	}
}

# The one way to open Agradent to a country. Anything may call it: it asks for nothing else.
# Without the Ambassador the first landing still has to be made, from the Agradent window.
DA_AG_unlock_travel = {
	custom_tooltip = DA_AG_unlock_travel_tt
	hidden_effect = {
		if = {
			limit = { NOT = { has_country_flag = DA_Agradent_travel_unlocked } }
			set_country_flag = DA_Agradent_travel_unlocked
			set_variable = { which = DA_AG_goodwill value = 50 }
			if = {
				limit = { NOT = { has_country_flag = DA_SZ_homecoming } }
				country_event = { id = DA_Agradent_event.1 days = 1 }
			}
		}
	}
}

# Goodwill of the canal folk: 0 to 100. The two events fire once each time the line is crossed.
DA_AG_goodwill_up = {
	custom_tooltip = DA_AG_goodwill_up_$amount$_tt
	hidden_effect = {
		change_variable = { which = DA_AG_goodwill value = $amount$ }
		DA_AG_check_goodwill = yes
	}
}

DA_AG_goodwill_down = {
	custom_tooltip = DA_AG_goodwill_down_$amount$_tt
	hidden_effect = {
		subtract_variable = { which = DA_AG_goodwill value = $amount$ }
		DA_AG_check_goodwill = yes
	}
}

DA_AG_check_goodwill = {
	if = {
		limit = { check_variable = { which = DA_AG_goodwill value = 100 } }
		set_variable = { which = DA_AG_goodwill value = 100 }
	}
	if = {
		limit = { NOT = { check_variable = { which = DA_AG_goodwill value = 0 } } }
		set_variable = { which = DA_AG_goodwill value = 0 }
	}
	if = {
		limit = {
			check_variable = { which = DA_AG_goodwill value = 85 }
			DA_AG_outposts_4 = yes
			NOT = { has_country_flag = DA_AG_waters_returned }
		}
		set_country_flag = DA_AG_waters_returned
		country_event = { id = DA_Agradent_event.40 days = 30 }
	}
	if = {
		limit = {
			NOT = { check_variable = { which = DA_AG_goodwill value = 15 } }
			NOT = { has_country_flag = DA_AG_locks_closed }
			DA_AG_any_outpost = yes
		}
		set_country_flag = DA_AG_locks_closed
		country_event = { id = DA_Agradent_event.41 days = 30 }
	}
	if = {
		limit = { has_country_flag = DA_AG_locks_closed DA_AG_resented = no }
		clr_country_flag = DA_AG_locks_closed
	}
}

# The Far Watch gives no goods: its outpost is a country modifier.
DA_AG_set_watch_modifier = {
	remove_country_modifier = DA_AG_far_watch_1
	remove_country_modifier = DA_AG_far_watch_2
	remove_country_modifier = DA_AG_far_watch_3
	if = {
		limit = { has_country_flag = DA_AG_outpost_watch_3 }
		add_country_modifier = { name = DA_AG_far_watch_3 duration = -1 }
	}
	else_if = {
		limit = { has_country_flag = DA_AG_outpost_watch_2 }
		add_country_modifier = { name = DA_AG_far_watch_2 duration = -1 }
	}
	else_if = {
		limit = { has_country_flag = DA_AG_outpost_watch_1 }
		add_country_modifier = { name = DA_AG_far_watch_1 duration = -1 }
	}
}

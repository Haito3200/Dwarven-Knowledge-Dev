# Construction uses the native queue: the engine charges once, applies cost/time modifiers and handles cancellation refunds.

DA_start_DA_Runeforged_fort = {
	if = {
		limit = { DA_can_construct_DA_Runeforged_fort = yes }
		add_building_construction = {
			building = DA_Runeforged_fort
			cost = 1
			speed = 1
		}
	}
}

# Save compatibility: retain the highest runic tier, otherwise magic, otherwise artificer.
DA_normalize_special_forts = {
	if = {
		limit = { has_building = DA_Runeforged_Amldihr_fort }
		DA_finish_DA_Runeforged_Amldihr_fort = yes
	}
	else_if = {
		limit = { has_building = DA_Hehogrim_Runeforged_fort }
		DA_finish_DA_Hehogrim_Runeforged_fort = yes
	}
	else_if = {
		limit = { has_building = DA_Runeforged_fort_t3 }
		DA_finish_DA_Runeforged_fort_t3 = yes
	}
	else_if = {
		limit = { has_building = DA_Runeforged_fort_t2 }
		DA_finish_DA_Runeforged_fort_t2 = yes
	}
	else_if = {
		limit = { has_building = DA_Runeforged_fort }
		DA_finish_DA_Runeforged_fort = yes
	}
	else_if = {
		limit = { has_building = fen_magic_fort_t3 }
		DA_finish_fen_magic_fort_t3 = yes
	}
	else_if = {
		limit = { has_building = fen_magic_fort_t2 }
		DA_finish_fen_magic_fort_t2 = yes
	}
	else_if = {
		limit = { has_building = fen_magic_fort_t1 }
		DA_finish_fen_magic_fort_t1 = yes
	}
	else_if = {
		limit = { has_building = fen_tech_fort_t3 }
		DA_finish_fen_tech_fort_t3 = yes
	}
	else_if = {
		limit = { has_building = fen_tech_fort_t2 }
		DA_finish_fen_tech_fort_t2 = yes
	}
	else_if = {
		limit = { has_building = fen_tech_fort_t1 }
		DA_finish_fen_tech_fort_t1 = yes
	}
}

# Country scope. Re-evaluated before each purchase; keep at least the build price in reserve.
DA_ai_build_one_special_fort = {
	random_owned_province = {
		limit = { DA_special_fort_ai_candidate = yes }
		if = {
			limit = {
				DA_can_construct_DA_Runeforged_Amldihr_fort = yes
				DA_can_afford_special_fort = { cost = 30000 minimum = 3000 }
			}
			DA_start_DA_Runeforged_Amldihr_fort = yes
		}
		else_if = {
			limit = {
				DA_can_construct_DA_Hehogrim_Runeforged_fort = yes
				DA_can_afford_special_fort = { cost = 6000 minimum = 600 }
			}
			DA_start_DA_Hehogrim_Runeforged_fort = yes
		}
		else_if = {
			limit = {
				DA_can_construct_DA_Runeforged_fort_t3 = yes
				DA_can_afford_special_fort = { cost = 3200 minimum = 320 }
			}
			DA_start_DA_Runeforged_fort_t3 = yes
		}
		else_if = {
			limit = {
				DA_can_construct_DA_Runeforged_fort_t2 = yes
				DA_can_afford_special_fort = { cost = 1500 minimum = 150 }
			}
			DA_start_DA_Runeforged_fort_t2 = yes
		}
		else_if = {
			limit = {
				DA_can_construct_DA_Runeforged_fort = yes
				DA_can_afford_special_fort = { cost = 500 minimum = 50 }
			}
			DA_start_DA_Runeforged_fort = yes
		}
		else_if = {
			limit = {
				DA_can_construct_fen_magic_fort_t3 = yes
				DA_can_afford_special_fort = { cost = 2400 minimum = 240 }
			}
			DA_start_fen_magic_fort_t3 = yes
		}
		else_if = {
			limit = {
				DA_can_construct_fen_magic_fort_t2 = yes
				DA_can_afford_special_fort = { cost = 1000 minimum = 100 }
			}
			DA_start_fen_magic_fort_t2 = yes
		}
		else_if = {
			limit = {
				DA_can_construct_fen_magic_fort_t1 = yes
				DA_can_afford_special_fort = { cost = 400 minimum = 40 }
			}
			DA_start_fen_magic_fort_t1 = yes
		}
		else_if = {
			limit = {
				DA_can_construct_fen_tech_fort_t3 = yes
				DA_can_afford_special_fort = { cost = 2400 minimum = 240 }
			}
			DA_start_fen_tech_fort_t3 = yes
		}
		else_if = {
			limit = {
				DA_can_construct_fen_tech_fort_t2 = yes
				DA_can_afford_special_fort = { cost = 1000 minimum = 100 }
			}
			DA_start_fen_tech_fort_t2 = yes
		}
		else_if = {
			limit = {
				DA_can_construct_fen_tech_fort_t1 = yes
				DA_can_afford_special_fort = { cost = 400 minimum = 40 }
			}
			DA_start_fen_tech_fort_t1 = yes
		}
	}
}

DA_ai_special_fortifications = {
	if = {
		limit = { ai = yes is_at_war = no NOT = { num_of_loans = 1 } }
		# At most three constructions per five-year pulse.
		DA_ai_build_one_special_fort = yes
		DA_ai_build_one_special_fort = yes
		DA_ai_build_one_special_fort = yes
	}
}

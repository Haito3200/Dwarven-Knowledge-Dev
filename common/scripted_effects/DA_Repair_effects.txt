# Option a of DA_Repair.1: free what stays locked. Nothing is lost.
DA_repair_locks = {
	custom_tooltip = DA_repair_locks_tt
	hidden_effect = {
		clr_country_flag = DA_rf_menu_in_use	# Research Facility menu
		clr_country_flag = DA_ADK_menu_in_use	# hold exploration menu
		clr_country_flag = DA_Buildings_menu_in_use	# buildings menu
		clr_country_flag = DA_alchemy_in_use	# potion crafter
		# a hold whose company came back from the deep but still counts as exploring
		every_owned_province = {
			limit = {
				has_province_flag = DA_expedition_in_progress
				NOT = { has_province_modifier = DA_HX_company_below }
			}
			clr_province_flag = DA_expedition_in_progress
		}
		DA_HX_refresh_ready_count = yes
		# research leftovers when nothing runs, or a research with no target (it could never end)
		if = {
			limit = {
				OR = {
					NOT = { has_country_flag = DA_research_in_progress }
					NOT = { check_variable = { which = DA_research_progress_MAX value = 0.001 } }
				}
			}
			DA_reset_research = yes
			remove_estate_privilege = DA_Runesmith_research_privilege	# -10% tax while a Runesmith Research runs
		}
	}
}

# Option b of DA_Repair.1: the research in progress ends; its progress is lost.
DA_repair_cancel_research = {
	custom_tooltip = DA_repair_cancel_research_tt
	hidden_effect = {
		DA_reset_research = yes
		remove_estate_privilege = DA_Runesmith_research_privilege
	}
}

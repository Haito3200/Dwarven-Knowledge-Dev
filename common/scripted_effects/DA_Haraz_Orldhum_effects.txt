# Warsuit Foundry of Haraz Orldhum (2868). Country scope. Written on 10 November 2026.

# Ends whichever warsuit doctrine is in force and records that we build warsuits of our own.
# The caller adds the new doctrine right after it: DA_HO_exoskeletons_as_reward (Suits of Honour),
# DA_HO_exoskeletons_in_pratical_military (The Platinum Vanguard) or DA_HO_exoskeletons_for_all_infantry (The Platinum Host).
# Called by DA_Haraz_Orldhum_event.3 (the first doctrine) and by the doctrine buttons of the wonder window.
DA_HO_end_doctrines = {
	hidden_effect = {
		remove_country_modifier = DA_HO_exoskeletons_as_reward
		remove_country_modifier = DA_HO_exoskeletons_in_pratical_military
		remove_country_modifier = DA_HO_exoskeletons_for_all_infantry
		set_country_flag = DA_HO_warsuits_in_production
	}
}

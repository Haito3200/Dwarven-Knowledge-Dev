# The Hidden Floor (.50): one more floor for the expedition. Anbennar brings the party home when clearedFloor reaches
# nbFloor (diggy_expedition.2000), so one floor less cleared is one more floor to go: more days, supplies and
# encounters, and one more floor of loot. Never below 0 (the party may have come home while the event was open).
DA_SP_add_hidden_floor = {
	custom_tooltip = DA_SP_hidden_floor_tt
	hidden_effect = {
		if = {
			limit = {
				check_variable = { clearedFloor = 1 }
			}
			subtract_variable = { clearedFloor = 1 }
		}
	}
}

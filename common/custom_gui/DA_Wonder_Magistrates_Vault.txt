# The works take time: the tier and what it brings arrive when DA_WW_finish_2914 runs
# (common/scripted_effects/DA_Wonder_Works_effects.txt). Materials are taken here, ducats by DA_WW_begin_2914.
custom_button = {
	name = DA_Magistrates_Vault_upgrade_button
	tooltip = DA_MV_upgrade_button_tt
	potential = {
		NOT = { has_country_flag = DA_WW_building_2914 }
		NOT = { DA_WW_maxed_2914 = yes }
	}
	trigger = {
		custom_trigger_tooltip = {
			tooltip = DA_Wonder_in_use_tt
			NOT = { has_country_flag = DA_Wonder_DA_Magistrates_Vault_in_use }
		}
		2914 = {
			owned_by = prev
			has_great_project = { type = DA_Magistrates_Vault tier = 0 }
		}
		if = {
			limit = { 2914 = { has_great_project = { type = DA_Magistrates_Vault tier = 0 } NOT = { has_great_project = { type = DA_Magistrates_Vault tier = 1 } } } }
			2914 = {
				custom_trigger_tooltip = {
					tooltip = DA_tooltip_has_dig_level_4_or_higher
					DA_has_dig_level_4_or_higher = yes
				}
			}
			DA_alchemy_reagents_total_required = { amount = 10 }
			DA_enchanted_building_material_total_required = { amount = 10 }
			DA_arcane_metal_total_required = { amount = 10 }
			DA_WW_can_pay_2914 = yes
		}
		if = {
			limit = { 2914 = { has_great_project = { type = DA_Magistrates_Vault tier = 1 } NOT = { has_great_project = { type = DA_Magistrates_Vault tier = 2 } } } }
			2914 = {
				custom_trigger_tooltip = {
					tooltip = DA_tooltip_has_dig_level_6_or_higher
					DA_has_dig_level_6_or_higher = yes
				}
			}
			2914 = { custom_trigger_tooltip = { tooltip = DA_MV_find2_tt has_province_flag = DA_Gulan_exp2 } }
			DA_alchemy_reagents_total_required = { amount = 20 }
			DA_enchanted_building_material_total_required = { amount = 20 }
			DA_arcane_metal_total_required = { amount = 20 }
			DA_WW_can_pay_2914 = yes
		}
		if = {
			limit = { 2914 = { has_great_project = { type = DA_Magistrates_Vault tier = 2 } NOT = { has_great_project = { type = DA_Magistrates_Vault tier = 3 } } } }
			2914 = {
				custom_trigger_tooltip = {
					tooltip = DA_tooltip_has_dig_level_8_or_higher
					DA_has_dig_level_8_or_higher = yes
				}
			}
			DA_alchemy_reagents_total_required = { amount = 40 }
			DA_enchanted_building_material_total_required = { amount = 40 }
			DA_arcane_metal_total_required = { amount = 40 }
			DA_WW_can_pay_2914 = yes
		}
	}
	effect = {
		hidden_effect = { DA_WW_refresh_2914 = yes }
		if = {
			limit = {
				2914 = { has_great_project = { type = DA_Magistrates_Vault tier = 0 } NOT = { has_great_project = { type = DA_Magistrates_Vault tier = 1 } } }
				DA_WW_can_pay_2914 = yes    # at the price of this very day
			}
			DA_alchemy_reagents_spend = { amount = 10 }
			DA_enchanted_building_material_spend = { amount = 10 }
			DA_arcane_metal_spend = { amount = 10 }
			DA_WW_begin_2914 = yes
			custom_tooltip = DA_MV_tier_1_unlock_tt
		}
		else_if = {
			limit = {
				2914 = { has_great_project = { type = DA_Magistrates_Vault tier = 1 } NOT = { has_great_project = { type = DA_Magistrates_Vault tier = 2 } } }
				DA_WW_can_pay_2914 = yes    # at the price of this very day
			}
			DA_alchemy_reagents_spend = { amount = 20 }
			DA_enchanted_building_material_spend = { amount = 20 }
			DA_arcane_metal_spend = { amount = 20 }
			DA_WW_begin_2914 = yes
			custom_tooltip = DA_MV_tier_2_unlock_tt
		}
		else_if = {
			limit = {
				2914 = { has_great_project = { type = DA_Magistrates_Vault tier = 2 } NOT = { has_great_project = { type = DA_Magistrates_Vault tier = 3 } } }
				DA_WW_can_pay_2914 = yes    # at the price of this very day
			}
			DA_alchemy_reagents_spend = { amount = 40 }
			DA_enchanted_building_material_spend = { amount = 40 }
			DA_arcane_metal_spend = { amount = 40 }
			DA_WW_begin_2914 = yes
			custom_tooltip = DA_MV_tier_3_unlock_tt
		}
	}
}

# The works take time: the tier and what it brings arrive when DA_WW_finish_4266 runs
# (common/scripted_effects/DA_Wonder_Works_effects.txt). Materials are taken here, ducats by DA_WW_begin_4266.
custom_button = {
	name = DA_Stramolgiv_upgrade_button
	tooltip = DA_ST_upgrade_button_tt
	potential = {
		NOT = { has_country_flag = DA_WW_building_4266 }
		NOT = { DA_WW_maxed_4266 = yes }
	}
	trigger = {
		custom_trigger_tooltip = {
			tooltip = DA_Wonder_in_use_tt
			NOT = { has_country_flag = DA_Wonder_DA_Stramolgiv_in_use }
		}
		4266 = {
			owned_by = prev
			has_great_project = { type = DA_Stramolgiv tier = 0 }
		}
		if = {
			limit = { 4266 = { has_great_project = { type = DA_Stramolgiv tier = 0 } NOT = { has_great_project = { type = DA_Stramolgiv tier = 1 } } } }
			4266 = {
				custom_trigger_tooltip = {
					tooltip = DA_tooltip_has_dig_level_4_or_higher
					DA_has_dig_level_4_or_higher = yes
				}
			}
			DA_alchemy_reagents_total_required = { amount = 10 }
			DA_enchanted_building_material_total_required = { amount = 10 }
			DA_arcane_metal_total_required = { amount = 10 }
			DA_WW_can_pay_4266 = yes
		}
		if = {
			limit = { 4266 = { has_great_project = { type = DA_Stramolgiv tier = 1 } NOT = { has_great_project = { type = DA_Stramolgiv tier = 2 } } } }
			4266 = {
				custom_trigger_tooltip = {
					tooltip = DA_tooltip_has_dig_level_6_or_higher
					DA_has_dig_level_6_or_higher = yes
				}
			}
			4266 = { custom_trigger_tooltip = { tooltip = DA_ST_find2_tt has_province_flag = DA_Krakazol_exp2 } }
			DA_alchemy_reagents_total_required = { amount = 20 }
			DA_enchanted_building_material_total_required = { amount = 20 }
			DA_arcane_metal_total_required = { amount = 20 }
			DA_WW_can_pay_4266 = yes
		}
		if = {
			limit = { 4266 = { has_great_project = { type = DA_Stramolgiv tier = 2 } NOT = { has_great_project = { type = DA_Stramolgiv tier = 3 } } } }
			4266 = {
				custom_trigger_tooltip = {
					tooltip = DA_tooltip_has_dig_level_8_or_higher
					DA_has_dig_level_8_or_higher = yes
				}
			}
			DA_alchemy_reagents_total_required = { amount = 40 }
			DA_enchanted_building_material_total_required = { amount = 40 }
			DA_arcane_metal_total_required = { amount = 40 }
			DA_WW_can_pay_4266 = yes
		}
	}
	effect = {
		hidden_effect = { DA_WW_refresh_4266 = yes }
		if = {
			limit = {
				4266 = { has_great_project = { type = DA_Stramolgiv tier = 0 } NOT = { has_great_project = { type = DA_Stramolgiv tier = 1 } } }
				DA_WW_can_pay_4266 = yes    # at the price of this very day
			}
			DA_alchemy_reagents_spend = { amount = 10 }
			DA_enchanted_building_material_spend = { amount = 10 }
			DA_arcane_metal_spend = { amount = 10 }
			DA_WW_begin_4266 = yes
			custom_tooltip = DA_ST_tier_1_unlock_tt
		}
		else_if = {
			limit = {
				4266 = { has_great_project = { type = DA_Stramolgiv tier = 1 } NOT = { has_great_project = { type = DA_Stramolgiv tier = 2 } } }
				DA_WW_can_pay_4266 = yes    # at the price of this very day
			}
			DA_alchemy_reagents_spend = { amount = 20 }
			DA_enchanted_building_material_spend = { amount = 20 }
			DA_arcane_metal_spend = { amount = 20 }
			DA_WW_begin_4266 = yes
			custom_tooltip = DA_ST_tier_2_unlock_tt
		}
		else_if = {
			limit = {
				4266 = { has_great_project = { type = DA_Stramolgiv tier = 2 } NOT = { has_great_project = { type = DA_Stramolgiv tier = 3 } } }
				DA_WW_can_pay_4266 = yes    # at the price of this very day
			}
			DA_alchemy_reagents_spend = { amount = 40 }
			DA_enchanted_building_material_spend = { amount = 40 }
			DA_arcane_metal_spend = { amount = 40 }
			DA_WW_begin_4266 = yes
			custom_tooltip = DA_ST_tier_3_unlock_tt
		}
	}
}

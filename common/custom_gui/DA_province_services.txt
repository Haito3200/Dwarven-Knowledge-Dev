# ROOT is the selected province; FROM is the viewing country.

custom_button = {
	name = DA_province_button_medicine
	tooltip = DA_province_button_medicine_tt
	potential = { always = yes }
	trigger = { owned_by = FROM }
	effect = {
		FROM = {
			clr_country_flag = DA_province_fortifications_open
			if = {
				limit = { has_country_flag = DA_province_medicine_open }
				clr_country_flag = DA_province_medicine_open
			}
			else = { set_country_flag = DA_province_medicine_open }
		}
	}
	frame = { number = 1 trigger = { has_province_modifier = DA_Medicine.Quarantined_province } }
	frame = { number = 3 trigger = { DA_has_disease_in_province = yes } }
	frame = { number = 4 trigger = { always = yes } }
}

	name = DA_province_medicine_panel

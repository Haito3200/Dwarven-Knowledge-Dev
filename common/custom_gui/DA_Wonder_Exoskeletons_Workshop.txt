# BEGIN DA_WW level-up button (build_wonder_pages.py)
# The works take time: the tier and what it brings arrive when DA_WW_finish_2868 runs
# (common/scripted_effects/DA_Wonder_Works_effects.txt). Materials are taken here, ducats by DA_WW_begin_2868.
custom_button = {
	name = DA_Exoskeletons_Workshop_upgrade_button
	tooltip = DA_HO_upgrade_button_tt
	potential = {
		NOT = { has_country_flag = DA_WW_building_2868 }
		NOT = { DA_WW_maxed_2868 = yes }
	}
	trigger = {
		custom_trigger_tooltip = {
			tooltip = DA_Wonder_in_use_tt
			NOT = { has_country_flag = DA_Wonder_DA_Exoskeletons_Workshop_in_use }
		}
		2868 = {
			owned_by = prev
			has_great_project = { type = DA_Exoskeletons_Workshop tier = 0 }
		}
		if = {
			limit = { 2868 = { has_great_project = { type = DA_Exoskeletons_Workshop tier = 0 } NOT = { has_great_project = { type = DA_Exoskeletons_Workshop tier = 1 } } } }
			2868 = {
				custom_trigger_tooltip = {
					tooltip = DA_tooltip_has_dig_level_4_or_higher
					DA_has_dig_level_4_or_higher = yes
				}
			}
			DA_alchemy_reagents_total_required = { amount = 10 }
			DA_enchanted_building_material_total_required = { amount = 10 }
			DA_arcane_metal_total_required = { amount = 10 }
			DA_WW_can_pay_2868 = yes
		}
		if = {
			limit = { 2868 = { has_great_project = { type = DA_Exoskeletons_Workshop tier = 1 } NOT = { has_great_project = { type = DA_Exoskeletons_Workshop tier = 2 } } } }
			2868 = {
				custom_trigger_tooltip = {
					tooltip = DA_tooltip_has_dig_level_6_or_higher
					DA_has_dig_level_6_or_higher = yes
				}
			}
			DA_alchemy_reagents_total_required = { amount = 20 }
			DA_enchanted_building_material_total_required = { amount = 20 }
			DA_arcane_metal_total_required = { amount = 20 }
			DA_WW_can_pay_2868 = yes
		}
		if = {
			limit = { 2868 = { has_great_project = { type = DA_Exoskeletons_Workshop tier = 2 } NOT = { has_great_project = { type = DA_Exoskeletons_Workshop tier = 3 } } } }
			2868 = {
				custom_trigger_tooltip = {
					tooltip = DA_tooltip_has_dig_level_8_or_higher
					DA_has_dig_level_8_or_higher = yes
				}
			}
			DA_alchemy_reagents_total_required = { amount = 40 }
			DA_enchanted_building_material_total_required = { amount = 40 }
			DA_arcane_metal_total_required = { amount = 40 }
			DA_WW_can_pay_2868 = yes
		}
	}
	effect = {
		hidden_effect = { DA_WW_refresh_2868 = yes }
		if = {
			limit = {
				2868 = { has_great_project = { type = DA_Exoskeletons_Workshop tier = 0 } NOT = { has_great_project = { type = DA_Exoskeletons_Workshop tier = 1 } } }
				DA_WW_can_pay_2868 = yes    # at the price of this very day
			}
			DA_alchemy_reagents_spend = { amount = 10 }
			DA_enchanted_building_material_spend = { amount = 10 }
			DA_arcane_metal_spend = { amount = 10 }
			DA_WW_begin_2868 = yes
			custom_tooltip = DA_HO_tier_1_unlock_tt
		}
		else_if = {
			limit = {
				2868 = { has_great_project = { type = DA_Exoskeletons_Workshop tier = 1 } NOT = { has_great_project = { type = DA_Exoskeletons_Workshop tier = 2 } } }
				DA_WW_can_pay_2868 = yes    # at the price of this very day
			}
			DA_alchemy_reagents_spend = { amount = 20 }
			DA_enchanted_building_material_spend = { amount = 20 }
			DA_arcane_metal_spend = { amount = 20 }
			DA_WW_begin_2868 = yes
			custom_tooltip = DA_HO_tier_2_unlock_tt
		}
		else_if = {
			limit = {
				2868 = { has_great_project = { type = DA_Exoskeletons_Workshop tier = 2 } NOT = { has_great_project = { type = DA_Exoskeletons_Workshop tier = 3 } } }
				DA_WW_can_pay_2868 = yes    # at the price of this very day
			}
			DA_alchemy_reagents_spend = { amount = 40 }
			DA_enchanted_building_material_spend = { amount = 40 }
			DA_arcane_metal_spend = { amount = 40 }
			DA_WW_begin_2868 = yes
			custom_tooltip = DA_HO_tier_3_unlock_tt
		}
	}
}

custom_button = { # doctrine 1: frame 2 while it is in force, frame 3 while it cannot be adopted
	name = DA_Exoskeletons_Workshop_button_doctrine_1
	tooltip = DA_HO_button_doctrine_1_tt
	potential = {
		always = yes
	}
	frame = {
		number = 2
		trigger = { has_country_modifier = DA_HO_exoskeletons_as_reward }
	}
	frame = {
		number = 3
		trigger = {
			OR = {
				NOT = { has_country_flag = DA_HO_warsuits_in_production }
				NOT = { DA_HO_foundry_at_tier_1 = yes }
			}
		}
	}
	frame = {
		number = 1
		trigger = { always = yes }
	}
	trigger = {
		custom_trigger_tooltip = {
			tooltip = DA_HO_production_started_tt
			has_country_flag = DA_HO_warsuits_in_production
		}
		custom_trigger_tooltip = {
			tooltip = DA_HO_needs_tier_1_tt
			DA_HO_foundry_at_tier_1 = yes
		}
		custom_trigger_tooltip = {
			tooltip = DA_HO_not_this_doctrine_tt
			NOT = { has_country_modifier = DA_HO_exoskeletons_as_reward }
		}
		custom_trigger_tooltip = {
			tooltip = DA_HO_doctrine_not_changed_tt
			NOT = { has_country_modifier = DA_HO_doctrine_CD }
		}
		custom_trigger_tooltip = {
			tooltip = DA_has_5_ADK
			check_variable = {
				which = ancientDwarvenKnowledge
				value = 5
			}
		}
		treasury = 1500
	}
	effect = {
		add_treasury = -1500
		custom_tooltip = DA_Cost_5_ADK
		DA_HO_end_doctrines = yes
		add_country_modifier = {
			name = DA_HO_exoskeletons_as_reward
			duration = -1
		}
		custom_tooltip = DA_HO_doctrine_changed_tt
		hidden_effect = {
			subtract_variable = { ancientDwarvenKnowledge = 5 }
			add_country_modifier = {
				name = DA_HO_doctrine_CD
				duration = 3650
			}
		}
	}
}

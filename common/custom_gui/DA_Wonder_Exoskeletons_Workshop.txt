# Warsuit Foundry, Haraz Orldhum (2868): the wonder window. Rewritten on 10 November 2026.
# The .gui side is the window "DA_Exoskeletons_Workshop_Window" in interface/topbar.gui.
# ROOT and FROM are the country that looks at the window.
#
# The upgrade button restores the foundry (tiers I to III). Each tier allows one more warsuit doctrine
# and one more tier of the Turrets of Orldhum (the turret button is in DA_provinceview.txt).
# The first doctrine is chosen in DA_Haraz_Orldhum_event.3, at the end of the Runesmith project
# DA_RS_project_Platinum_exoskeletons; after that the three doctrine buttons change it
# (1500 ducats, 5 Ancient Dwarven Knowledge, once every 10 years).
# The story is in events/DA_Hold_Haraz_Orldhum.txt.

custom_button = { # restores the foundry, tier by tier; each tier unlocks a doctrine and a tier of the turrets
	name = DA_Exoskeletons_Workshop_upgrade_button
	tooltip = DA_HO_upgrade_button_tt
	potential = {
		always = yes
	}
	trigger = {
		custom_trigger_tooltip = {
			tooltip = DA_Wonder_in_use_tt
			NOT = { has_country_flag = DA_Wonder_DA_Exoskeletons_Workshop_in_use }
		}
		2868 = {
			owned_by = prev
			has_great_project = {
				type = DA_Exoskeletons_Workshop
				tier = 0
			}
		}
		if = {
			limit = {
				2868 = {
					has_great_project = {
						type = DA_Exoskeletons_Workshop
						tier = 0
					}
					NOT = { has_great_project = { type = DA_Exoskeletons_Workshop tier = 1 } }
				}
			}
			2868 = {
				custom_trigger_tooltip = {
					tooltip = DA_tooltip_has_dig_level_4_or_higher
					DA_has_dig_level_4_or_higher = yes
				}
			}
			DA_alchemy_reagents_total_required = {
				amount = 10
			}
			DA_enchanted_building_material_total_required = {
				amount = 10
			}
			DA_arcane_metal_total_required = {
				amount = 10
			}
			treasury = 1600
		}
		if = {
			limit = {
				2868 = {
					has_great_project = {
						type = DA_Exoskeletons_Workshop
						tier = 1
					}
					NOT = { has_great_project = { type = DA_Exoskeletons_Workshop tier = 2 } }

			}
			2868 = {

				amount = 40
			}
			treasury = 7500
		}
		if = {
			limit = {
				2868 = {
					has_great_project = {
						type = DA_Exoskeletons_Workshop
						tier = 3
					}
				}
			}
			custom_trigger_tooltip = {
				tooltip = DA_Maxed_wonder_tt
				always = no
			}
		}
	}
	effect = {
		if = {
			limit = {
				2868 = {
					has_great_project = {
						type = DA_Exoskeletons_Workshop
						tier = 0
					}
					NOT = { has_great_project = { type = DA_Exoskeletons_Workshop tier = 1 } }
				}
			}
			DA_alchemy_reagents_spend = {
				amount = 10
			}
			DA_enchanted_building_material_spend = {
				amount = 10
			}
			DA_arcane_metal_spend = {
				amount = 10
			}
			add_treasury = -1600
			2868 = {

					type = DA_Exoskeletons_Workshop
					tier = 1
				}
			}
			custom_tooltip = DA_HO_tier_1_unlock_tt
			hidden_effect = {
				set_country_flag = DA_can_implement_platinum_exoskeletons # the Runesmith project Platinum Warsuits
				country_event = { id = DA_Haraz_Orldhum_event.2 days = 30 }
				country_event = { id = DA_Haraz_Orldhum_event.21 days = 365 }
				country_event = { id = DA_Haraz_Orldhum_event.22 days = 1095 }
			}
		}
		else_if = {

					}

			}

			}

			}

					tier = 1
				}
			}
			custom_tooltip = DA_HO_tier_2_unlock_tt
			hidden_effect = {
				country_event = { id = DA_Haraz_Orldhum_event.4 days = 30 }
			}

custom_button = { # doctrine 1: frame 2 while it is in force, frame 3 while it cannot be adopted
	name = DA_Exoskeletons_Workshop_button_doctrine_1
	tooltip = DA_HO_button_doctrine_1_tt
	potential = {
		always = yes
	}
	frame = {
		number = 2
		trigger = { has_country_modifier = DA_HO_exoskeletons_as_reward }
	}
	frame = {
		number = 3
		trigger = {
			OR = {
				NOT = { has_country_flag = DA_HO_warsuits_in_production }
				NOT = { DA_HO_foundry_at_tier_1 = yes }
			}
		}
	}
	frame = {
		number = 1
		trigger = { always = yes }
	}
	trigger = {
		custom_trigger_tooltip = {
			tooltip = DA_HO_production_started_tt
			has_country_flag = DA_HO_warsuits_in_production
		}
		custom_trigger_tooltip = {
			tooltip = DA_HO_needs_tier_1_tt
			DA_HO_foundry_at_tier_1 = yes
		}
		custom_trigger_tooltip = {
			tooltip = DA_HO_not_this_doctrine_tt
			NOT = { has_country_modifier = DA_HO_exoskeletons_as_reward }
		}
		custom_trigger_tooltip = {
			tooltip = DA_HO_doctrine_not_changed_tt
			NOT = { has_country_modifier = DA_HO_doctrine_CD }
		}
		custom_trigger_tooltip = {
			tooltip = DA_has_5_ADK
			check_variable = {
				which = ancientDwarvenKnowledge
				value = 5
			}
		}
		treasury = 1500
	}
	effect = {
		add_treasury = -1500
		custom_tooltip = DA_Cost_5_ADK
		DA_HO_end_doctrines = yes
		add_country_modifier = {
			name = DA_HO_exoskeletons_as_reward
			duration = -1
		}
		custom_tooltip = DA_HO_doctrine_changed_tt
		hidden_effect = {
			subtract_variable = { ancientDwarvenKnowledge = 5 }
			add_country_modifier = {
				name = DA_HO_doctrine_CD
				duration = 3650
			}
		}
	}
}

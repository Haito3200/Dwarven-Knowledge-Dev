# Agradent: landing sites, pods and outposts. Written on 10 November 2026.
# The .gui side is the window "DA_Agradent_Window" in interface/topbar.gui; it opens in the Dwarovar Wonders screen
# (selector button 23, or the Agradent button of the Setzunkhupa window).
# ROOT and FROM are the country that looks at the window. It asks only for DA_Agradent_travel_unlocked.
#
# Click a site on the map to select it. Launch a Pod throws an anchor at the selected site (the dice are rolled
# in DA_Agradent_event.11 to .17, 540 days later). On an anchored site the second button raises and enlarges the outpost.
# Outposts send their yield once a year (DA_AG_yearly_yield). Events are in events/DA_Agradent_events.txt.

custom_button = { # throws an anchor at the selected site
	name = DA_Agradent_button_pod
	tooltip = DA_AG_button_pod_tt
	potential = {
		always = yes
	}
	frame = { number = 2 trigger = { DA_AG_selected_pod_flying = yes } }
	frame = { number = 3 trigger = { OR = { NOT = { DA_AG_any_selected = yes } DA_AG_selected_anchored = yes has_country_flag = DA_AG_pod_in_flight } } }
	frame = { number = 1 trigger = { always = yes } }
	trigger = {
		custom_trigger_tooltip = {
			tooltip = DA_AG_site_selected_tt
			DA_AG_any_selected = yes
		}
		custom_trigger_tooltip = {
			tooltip = DA_AG_no_anchor_there_tt
			NOT = { DA_AG_selected_anchored = yes }
		}
		custom_trigger_tooltip = {
			tooltip = DA_AG_no_pod_flying_tt
			NOT = { has_country_flag = DA_AG_pod_in_flight }
		}
		DA_alchemy_reagents_total_required = {
			amount = 10
		}
		DA_enchanted_building_material_total_required = {
			amount = 15
		}
		DA_arcane_metal_total_required = {
			amount = 15
		}
		treasury = 1500
	}
	effect = {
		add_treasury = -1500
		DA_alchemy_reagents_spend = {
			amount = 10
		}
		DA_enchanted_building_material_spend = {
			amount = 15
		}
		DA_arcane_metal_spend = {
			amount = 15
		}
		custom_tooltip = DA_AG_pod_launched_tt
		hidden_effect = {
			set_country_flag = DA_AG_pod_in_flight
			if = {
				limit = { has_country_flag = DA_AG_sel_canal }
				set_country_flag = DA_AG_pod_canal
				country_event = { id = DA_Agradent_event.11 days = 540 }
			}
			else_if = {
				limit = { has_country_flag = DA_AG_sel_sea }
				set_country_flag = DA_AG_pod_sea
				country_event = { id = DA_Agradent_event.12 days = 540 }
			}
			else_if = {
				limit = { has_country_flag = DA_AG_sel_uplands }
				set_country_flag = DA_AG_pod_uplands
				country_event = { id = DA_Agradent_event.13 days = 540 }
			}
			else_if = {
				limit = { has_country_flag = DA_AG_sel_locks }
				set_country_flag = DA_AG_pod_locks
				country_event = { id = DA_Agradent_event.14 days = 540 }
			}
			else_if = {
				limit = { has_country_flag = DA_AG_sel_ice }
				set_country_flag = DA_AG_pod_ice
				country_event = { id = DA_Agradent_event.15 days = 540 }
			}
			else_if = {
				limit = { has_country_flag = DA_AG_sel_ringfall }
				set_country_flag = DA_AG_pod_ringfall
				country_event = { id = DA_Agradent_event.16 days = 540 }
			}
			else_if = {
				limit = { has_country_flag = DA_AG_sel_watch }
				set_country_flag = DA_AG_pod_watch
				country_event = { id = DA_Agradent_event.17 days = 540 }
			}
		}
	}
}

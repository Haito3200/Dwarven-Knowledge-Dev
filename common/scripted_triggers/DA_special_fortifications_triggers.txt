# Province-scoped rules shared by buildings, province buttons and the five-year AI pulse.

	OR = {
		has_building = DA_Runeforged_fort
		has_building = DA_Runeforged_fort_t2
		has_building = DA_Runeforged_fort_t3
		has_building = DA_Hehogrim_Runeforged_fort
		has_building = DA_Runeforged_Amldihr_fort

	custom_trigger_tooltip = {
		tooltip = DA_special_fort_standard_fort_tt

			has_building = fort_15th

# A lower tier in the same family may be replaced; other families may never coexist.
DA_special_fort_slot_available = {
	custom_trigger_tooltip = {
		tooltip = DA_special_fort_slot_available_tt
		OR = {
			DA_has_special_fort = no
			DA_has_$family$_fort = yes
		}
		NOT = { AND = { DA_has_runic_fort = yes DA_has_magic_fort = yes } }
		NOT = { AND = { DA_has_runic_fort = yes DA_has_tech_fort = yes } }
		NOT = { AND = { DA_has_magic_fort = yes DA_has_tech_fort = yes } }
	}
}

DA_special_fort_construction_site = {
	custom_trigger_tooltip = {
		tooltip = DA_special_fort_secure_city_tt
		is_city = yes
		controlled_by = owner
		has_siege = no
	}
	custom_trigger_tooltip = {
		tooltip = DA_special_fort_idle_province_tt
		NOT = { has_construction = any }
		NOT = { has_colonist = yes }
		NOT = { has_missionary = yes }
	}
	custom_trigger_tooltip = {
		tooltip = DA_special_fort_building_space_tt
		OR = {
			num_free_building_slots = 1
			DA_has_special_fort = yes
		}
	}
}

# Mirrors the engine's additive global/local build-cost modifiers and 10% floor.
# Arithmetic-trigger variables are temporary; prices and treasury are checked live.
DA_can_afford_special_fort = {
	variable_arithmetic_trigger = {
		custom_tooltip = DA_special_fort_can_afford_$cost$_tt
		export_to_variable = { which = DA_sf_global_cost value = modifier:build_cost who = owner }
		export_to_variable = { which = DA_sf_cost value = modifier:local_build_cost }
		export_to_variable = { which = DA_sf_gold value = treasury who = owner }
		change_variable = { which = DA_sf_cost which = DA_sf_global_cost }
		change_variable = { which = DA_sf_cost value = 1 }
		multiply_variable = { which = DA_sf_cost value = $cost$ }
		OR = {
			AND = {
				check_variable = { which = DA_sf_cost value = $minimum$ }
				check_variable = { which = DA_sf_gold which = DA_sf_cost }
			}
			AND = {
				NOT = { check_variable = { which = DA_sf_cost value = $minimum$ } }
				check_variable = { which = DA_sf_gold value = $minimum$ }
			}
		}
	}
}

		tooltip = DA_special_fort_mil_tech_1_tt

# Only established forts at strategically useful sites; no loans or wartime spending.
DA_special_fort_ai_site = {
	DA_special_fort_construction_site = yes
	DA_has_standard_fort = yes
	OR = {
		is_capital = yes
		has_terrain = dwarven_hold
		has_terrain = dwarven_hold_surface
		any_neighbor_province = {
			has_owner = yes
			NOT = { owned_by = PREV }
		}
	}
	owner = {
		ai = yes
		is_at_war = no
		NOT = { num_of_loans = 1 }
	}
}

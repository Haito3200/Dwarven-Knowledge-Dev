# Founding costs follow the site's minimum technology, not the owner's highest tier.
DA_mining_site_tier_1 = { OR = { DA_mining_site_hills = yes DA_mining_site_open = yes } }
DA_mining_site_tier_2 = { OR = { DA_mining_site_lowlands = yes DA_mining_site_wooded = yes } }
DA_mining_site_tier_3 = { OR = { DA_mining_site_road = yes DA_mining_site_dense = yes } }
DA_mining_site_tier_4 = { DA_mining_site_extreme = yes }
DA_mining_terrain_unlocked = {
 OR = {
  AND = { DA_mining_site_tier_1 = yes owner = { has_country_flag = DA_mining_technology_1 } }
  AND = { DA_mining_site_tier_2 = yes owner = { has_country_flag = DA_mining_technology_2 } }
  AND = { DA_mining_site_tier_3 = yes owner = { has_country_flag = DA_mining_technology_3 } }
  AND = { DA_mining_site_tier_4 = yes owner = { has_country_flag = DA_mining_technology_4 } }
 }
}
DA_mining_can_afford = {
 OR = {
  AND = { DA_mining_site_tier_1 = yes owner = { adm_power = 15 dip_power = 15 mil_power = 15 treasury = 25 } }
  AND = { DA_mining_site_tier_2 = yes owner = { adm_power = 25 dip_power = 25 mil_power = 25 treasury = 50 } }
  AND = { DA_mining_site_tier_3 = yes owner = { adm_power = 35 dip_power = 35 mil_power = 35 treasury = 75 } }
  AND = { DA_mining_site_tier_4 = yes owner = { adm_power = 50 dip_power = 50 mil_power = 50 treasury = 100 } }
 }
}

# The cutoff is target - 5, calculated at research start and when loading a save.
# This is only checked when selecting an event, not by the monthly research tick.
DA_can_receive_random_research_event = {
	has_country_flag = DA_research_in_progress
	check_variable = { which = DA_research_progress_MAX value = 0.001 }
	NOT = {
		check_variable = {
			which = DA_research_progress
			which = DA_research_event_cutoff
		}
	}
	# 17 November 2026: never during a school study, nor while the slot studies a school level
	NOT = { DA_SCH_studying_any = yes }
	NOT = { DA_research_is_school_research = yes }
}

# An open popup belongs to one research run, even if the same project is restarted.
# Each event ID has its own snapshot and cannot open twice before being answered.
DA_research_event_is_current = {
	has_country_flag = DA_research_in_progress
	has_country_flag = DA_research_event_pending.$id$
	check_variable = { which = DA_research_progress_MAX value = 0.001 }
	is_variable_equal = {
		which = DA_research_generation
		which = DA_research_event_generation.$id$
	}
	NOT = {
		check_variable = {
			which = DA_research_progress
			which = DA_research_progress_MAX
		}
	}
}

# The slot is studying a school level (Projects & Research, Miscellaneous Research): no random event then.
DA_research_is_school_research = {
	OR = {
		has_country_flag = DA_research_flag.alchemy
		has_country_flag = DA_research_flag.arcane_arts
		has_country_flag = DA_research_flag.mechanics
		has_country_flag = DA_research_flag.medicine
		has_country_flag = DA_research_flag.metallurgy
	}
}

# An event of the yearly roll is ready: research running, no popup of this id open, not seen in the last $days$ days.
DA_research_event_ready = {
	DA_can_receive_random_research_event = yes
	NOT = { has_country_flag = DA_research_event_pending.$id$ }
	OR = {
		NOT = { has_country_flag = DA_research_event_seen.$id$ }
		had_country_flag = { flag = DA_research_event_seen.$id$ days = $days$ }
	}
}

# A people of race $race$ works in our research: the race of our primary culture, or a people we rule that we hold at
# least coexisting (Anbennar's tolerance) and that is the majority in at least 3 of our provinces.
DA_research_race_people = {
	OR = {
		culture_is_$race$ = yes
		AND = {
			OR = {
				medium_tolerance_$race$_race_trigger = yes
				high_tolerance_$race$_race_trigger = yes
			}
			num_of_owned_provinces_with = {
				value = 3
				has_$race$_majority_trigger = yes
			}
		}
	}
}

# A race event (DA_Research_events.31-.50) is ready.
DA_research_race_ready = {
	DA_research_race_people = { race = $race$ }
	DA_research_event_ready = { id = $id$ days = 10950 }
}

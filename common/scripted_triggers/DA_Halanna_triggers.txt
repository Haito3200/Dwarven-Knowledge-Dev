# Ancient Research Facility of Amldihr (4119): the Abyssal Invasion, Halanna's Heart and her recovery.
# Country scope unless noted. Written on 10 November 2026.

# The three states of the wonder window
DA_ARF_state_sealed = {
	NOT = { has_country_flag = DA_Halanna_awake }
	4119 = {
		has_province_flag = DA_Runic_gate_closed_p
		NOT = { has_province_flag = DA_Halanna_heart_found }
	}
}
DA_ARF_state_heart = {
	NOT = { has_country_flag = DA_Halanna_awake }
	4119 = { has_province_flag = DA_Halanna_heart_found }
}
DA_ARF_state_mother = {
	has_country_flag = DA_Halanna_awake
}

# The window can be opened from the Dwarovar Wonders selector
DA_ARF_window_available = {
	OR = {
		has_country_flag = DA_Halanna_awake
		4119 = {
			OR = {
				has_province_flag = DA_Runic_gate_closed_p
				has_province_flag = DA_Halanna_heart_found
				has_great_project = {
					type = DA_Ancient_Research_facility1
				}
			}
		}
	}
}

# Hold development with the 'don't spend ADK' toggle on (DA_Disabled_ADK_on_Click).
# Anbennar's anb_on_development_effect spends 1 ADK for +1 development in holds.
# This refund must run BEFORE Anbennar's hook: if it ran after, a country with
# exactly 1 ADK would already be at 0 and the refund would be skipped.
# The 000_ prefix sorts this file before Anbennar's 00_on_actions.txt.

on_adm_development = {
	if = {
		limit = {
			OR = {
				has_terrain = dwarven_hold
				has_terrain = dwarven_hold_surface
			}
			owner = {
				has_country_flag = DA_Disabled_ADK_on_Click
				check_variable = {
					which = ancientDwarvenKnowledge
					value = 1
				}
			}
		}
		add_base_tax = -1
		owner = {
			change_variable = {
				which = ancientDwarvenKnowledge
				value = 1
			}
		}
		add_next_institution_embracement_scaled = -5
	}
}

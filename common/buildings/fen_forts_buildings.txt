# Integrated Fen's Forts. No technology unlocks: construction is scripted from the province panel or AI pulse.

fen_magic_fort_t1 = {
	cost = 200
	time = 24
	onmap = no
	show_separate = yes
	influencing_fort = yes

	modifier = {
		fort_level = 1
		local_defensiveness = 0.10
	}

	build_trigger = {
		DA_requirements_fen_magic_fort_t1 = yes
	}

	on_built = {
		DA_finish_fen_magic_fort_t1 = yes
	}
	on_destroyed = { }
	on_construction_started = { }
	on_construction_canceled = { }

	keep_trigger = {
		owner = {
			at_least_magic_project_level = { project = magical_fortress level = 1 }
		}
	}

	ai_will_do = {
		factor = 0
	}
}

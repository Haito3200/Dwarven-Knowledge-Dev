# Integrated Fen's Forts. No technology unlocks: construction is scripted from the province panel or AI pulse.
# No make_obsolete either: with it EU4 crashed when a built fort was hovered. Each tier's on_built
# (DA_finish_<building>) removes the other special forts instead.

fen_magic_fort_t1 = {
	cost = 200
	time = 24
	onmap = no
	show_separate = yes
	influencing_fort = yes

	modifier = {
		fort_level = 1
		local_defensiveness = 0.10
	}

	build_trigger = {
		DA_requirements_fen_magic_fort_t1 = yes
	}

	on_built = {
		DA_finish_fen_magic_fort_t1 = yes
		DA_MEC_sync_fortif = yes   # DA_MEC (doc 83): its Mechanics upgrade (build_mechanics_content.py)
	}
	on_destroyed = { }
	on_construction_started = { }
	on_construction_canceled = { }

	keep_trigger = {
		owner = {
			at_least_magic_project_level = { project = magical_fortress level = 1 }
		}
	}

	ai_will_do = {
		factor = 0
	}
}

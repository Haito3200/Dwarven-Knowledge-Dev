province_event = { # a rising: every 60 months of quiet while the king lives (DA_VK_monthly counts them)
	id = DA_Verkal_Kozenad_event.3
	title = "DA_Verkal_Kozenad_event.3.t"
	desc = "DA_Verkal_Kozenad_event.3.d"
	picture = UNDEAD_ARMY_eventPicture

	is_triggered_only = yes
	trigger = {
		has_province_modifier = DA_continous_Undeads
		NOT = { has_province_flag = DA_Verkal_koz_king_dead }
		NOT = { has_province_modifier = DA_LE_contained_lich }	# the palace is sealed (Legendary Encounter window)
	}

	option = { # the size follows the dig level of the hold
		name = "DA_Verkal_Kozenad_event.3.a"
		ai_chance = { factor = 100 }
		if = {
			limit = { DA_has_dig_level_9_or_higher = yes }
			spawn_rebels = {
				type = DA_Undead_rebels
				size = 4
			}
		}
		if = {
			limit = { DA_has_dig_level_7_or_higher = yes NOT = { DA_has_dig_level_9_or_higher = yes } }
			spawn_rebels = {
				type = DA_Undead_rebels
				size = 3
			}
		}
		if = {
			limit = { NOT = { DA_has_dig_level_7_or_higher = yes } }
			spawn_rebels = {
				type = DA_Undead_rebels
				size = 2
			}
		}
	}
	option = {
		name = "DA_Verkal_Kozenad_event.3.b"
		ai_chance = { factor = 0 }
		trigger = {
			OR = {
				AND = { DA_has_dig_level_9_or_higher = yes owner = { treasury = 250 } }
				AND = { DA_has_dig_level_7_or_higher = yes NOT = { DA_has_dig_level_9_or_higher = yes } owner = { treasury = 175 } }
				AND = { NOT = { DA_has_dig_level_7_or_higher = yes } owner = { treasury = 100 } }
			}
		}
		if = {
			limit = { DA_has_dig_level_9_or_higher = yes }
			owner = { add_treasury = -250 }
		}
		if = {
			limit = { DA_has_dig_level_7_or_higher = yes NOT = { DA_has_dig_level_9_or_higher = yes } }
			owner = { add_treasury = -175 }
		}
		if = {
			limit = { NOT = { DA_has_dig_level_7_or_higher = yes } }
			owner = { add_treasury = -100 }
		}
		add_province_modifier = {
			name = DA_VK_galleries_walled
			duration = 730
		}
		custom_tooltip = DA_Verkal_Kozenad_event.3.b.tt
	}
}

country_event = { # one year into the project
	id = DA_Orlazam_az_dihr_event.11
	title = "DA_Orlazam_az_dihr_event.11.t"
	desc = "DA_Orlazam_az_dihr_event.11.d"
	picture = DA_Ramsteel_Forges_eventPicture

	is_triggered_only = yes
	trigger = {
		has_estate_privilege = DA_RS_project_Ramsteel_Forges
		NOT = { has_country_flag = DA_OAD_forge_midway }
	}

	immediate = {
		hidden_effect = {
			set_country_flag = DA_OAD_forge_midway
		}
	}

	option = { # cap the horns
		name = "DA_Orlazam_az_dihr_event.11.a"
		add_treasury = -200
		custom_tooltip = DA_Orlazam_az_dihr_event.11.a.tt
		hidden_effect = {
			set_country_flag = DA_OAD_forge_capped
		}
		ai_chance = {
			factor = 60
			modifier = {
				factor = 0
				NOT = { treasury = 300 }
			}
		}
	}
	option = { # let them harden
		name = "DA_Orlazam_az_dihr_event.11.b"
		add_estate_loyalty = {
			estate = estate_da_runesmiths
			loyalty = -5
		}
		ai_chance = {
			factor = 40
		}
	}
}

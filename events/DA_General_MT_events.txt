country_event = {
	id = DA_General_MT_event.4
	title = DA_General_MT_event.4.t
	desc = DA_General_MT_event.4.d
	picture = INVENTION_eventPicture

	is_triggered_only = yes

	immediate = {
		hidden_effect = { if = { limit = { capital_scope = { OR = { has_building = DA_Smelting_Furnace_1 has_building = DA_Smelting_Furnace_2 has_building = DA_Smelting_Furnace_3 } has_province_modifier = DA_central_mine } } capital_scope = { save_event_target_as = DA_MT_furnace_province } } if = { limit = { NOT = { capital_scope = { OR = { has_building = DA_Smelting_Furnace_1 has_building = DA_Smelting_Furnace_2 has_building = DA_Smelting_Furnace_3 } has_province_modifier = DA_central_mine } } any_core_province = { OR = { has_building = DA_Smelting_Furnace_1 has_building = DA_Smelting_Furnace_2 has_building = DA_Smelting_Furnace_3 } has_province_modifier = DA_central_mine } } random_core_province = { limit = { OR = { has_building = DA_Smelting_Furnace_1 has_building = DA_Smelting_Furnace_2 has_building = DA_Smelting_Furnace_3 } has_province_modifier = DA_central_mine } save_event_target_as = DA_MT_furnace_province } } if = { limit = { NOT = { capital_scope = { OR = { has_building = DA_Smelting_Furnace_1 has_building = DA_Smelting_Furnace_2 has_building = DA_Smelting_Furnace_3 } has_province_modifier = DA_central_mine } } NOT = { any_core_province = { OR = { has_building = DA_Smelting_Furnace_1 has_building = DA_Smelting_Furnace_2 has_building = DA_Smelting_Furnace_3 } has_province_modifier = DA_central_mine } } } random_core_province = { limit = { OR = { has_building = DA_Smelting_Furnace_1 has_building = DA_Smelting_Furnace_2 has_building = DA_Smelting_Furnace_3 } } save_event_target_as = DA_MT_furnace_province } } }
	}
	option = {
		name = "DA_General_MT_event.4.a"
		goto = DA_MT_furnace_province
		add_treasury = -500
		add_manpower = -10
		if = {
			limit = {
				capital_scope = {
					OR = {
						has_building = DA_Smelting_Furnace_1
						has_building = DA_Smelting_Furnace_2
						has_building = DA_Smelting_Furnace_3
					}
					has_province_modifier = DA_central_mine
				}
			}
			event_target:DA_MT_furnace_province = {
				random_list = {
					3 = { DA_add_Metal_Rare = yes } 
					1 = { DA_add_Metal_Legendary = yes } 
				}
			}
		}
		else_if = {
			limit = {
				any_core_province = {
					OR = {
						has_building = DA_Smelting_Furnace_1
						has_building = DA_Smelting_Furnace_2
						has_building = DA_Smelting_Furnace_3
					}
					has_province_modifier = DA_central_mine
				}
			}
			event_target:DA_MT_furnace_province = {
				random_list = {
					4 = { DA_add_Metal_Rare = yes } 
					1 = { DA_add_Metal_Legendary = yes } 
				}
			}
		}
		else = {
			event_target:DA_MT_furnace_province = {
				random_list = {
					5 = { DA_add_Metal_Rare = yes } 
					1 = { DA_add_Metal_Legendary = yes } 
				}
			}
		}
	}
}

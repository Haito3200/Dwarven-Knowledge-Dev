# Random research events. One roll a year (on_yearly_pulse_4 in common/on_actions/DA_on_actions.txt) while a
# research runs in the shared slot. Never during a school study, never while the slot studies a school level
# (the Miscellaneous Research button), never in the last 5 points of the target, never for the AI.
#   .1-.4, .6-.21   the yearly roll; each one at most once in 10 years
#   .5              The Shortcut Shows: follow-up of .8, for the same research only
#   .30             hidden: picks one race event
#   .31-.50         one event per Anbennar race; each one at most once in 30 years
# Every option acts only while the research its popup belongs to still runs (DA_research_event_is_current).
# Written 17 November 2026 by "Claude files/tools/apply_research_events_2026-11-17.py" (data and generator in
# "Claude files/tools/research_events_2026-11-17/"). To change a text or a number now, edit this file and
# localisation/DA_Research_events_l_english.yml.

country_event = {
	id = DA_Research_events.1		# A Request for Funds (yearly roll, weight 100)
	title = DA_Research_events.1.t
	desc = DA_Research_events.1.d
	picture = COURT_eventPicture
	is_triggered_only = yes

	trigger = {
		ai = no
		DA_research_event_ready = { id = 1 days = 3650 }
	}

	immediate = {
		DA_capture_research_event = { id = 1 }
		DA_research_event_seen = { id = 1 }
	}

	option = {		# The crown will pay.
		name = DA_Research_events.1.a
		trigger = {
			years_of_income = 0.25
		}
		if = {
			limit = { DA_research_event_is_current = { id = 1 } }
			add_years_of_income = -0.25
			DA_Research_add = { amount = 12 }
		}
		else = { custom_tooltip = DA_research_event_expired_tt }
	}
	option = {		# Let the guilds pay.
		name = DA_Research_events.1.b
		trigger = {
			has_estate = estate_burghers
		}
		if = {
			limit = { DA_research_event_is_current = { id = 1 } }
			DA_Research_add = { amount = 8 }
			add_estate_loyalty = { estate = estate_burghers loyalty = 5 }
			add_estate_influence_modifier = { estate = estate_burghers desc = DA_research_patrons_infl influence = 10 duration = 3650 }
		}
		else = { custom_tooltip = DA_research_event_expired_tt }
	}
	option = {		# Let the mage towers pay.

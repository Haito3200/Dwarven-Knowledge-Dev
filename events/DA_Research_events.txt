
namespace = DA_Research_events

# these are all flavor events which may happen while research is in progress, remember to put them in an yearly/bi_yearly on_action
# let's start with 10

country_event = {
	id = DA_Research_events.1
	title = DA_Research_events.1.t
	desc = DA_Research_events.1.d
	picture = INVENTION_eventPicture

	trigger = {
		ai = no
		DA_can_receive_random_research_event = yes
		NOT = { has_country_flag = DA_research_event_pending.1 }
	}
	mean_time_to_happen = {
		days = 1
	}

	is_triggered_only = yes

	immediate = {
		DA_capture_research_event = { id = 1 }
	}

	option = {
		name = DA_Research_events.1.a
		if = {
			limit = { DA_research_event_is_current = { id = 1 } }
			add_treasury = -250
			DA_Research_add = { amount = 6 }
		}
		else = {
			custom_tooltip = DA_research_event_expired_tt
		}
	}
	option = {
		name = DA_Research_events.1.b
		if = {
			limit = { DA_research_event_is_current = { id = 1 } }
			DA_Research_lose = { amount = 3 }
		}
		else = {
			custom_tooltip = DA_research_event_expired_tt
		}
	}
	after = {
		clr_country_flag = DA_research_event_pending.1
	}
}

	trigger = {
		ai = no
		DA_can_receive_random_research_event = yes
		NOT = { has_country_flag = DA_research_event_pending.9 }
		hidden_trigger = {
			NOT = { DA_unstoppable_research_in_progress = yes }
		}
	}
	mean_time_to_happen = {
		days = 1
	}

	is_triggered_only = yes

	immediate = {
		DA_capture_research_event = { id = 9 }
	}

	option = {
		name = DA_Research_events.9.a
		if = {
			limit = { DA_research_event_is_current = { id = 9 } }
			DA_Research_reset = yes
		}
		else = {
			custom_tooltip = DA_research_event_expired_tt
		}
	}
	option = {
		name = DA_Research_events.9.b
		highlight = yes
		trigger = {
			custom_trigger_tooltip = {
				tooltip = DA_has_10_ADK
				check_variable = {
					which = ancientDwarvenKnowledge
					value = 10
				}
			}
		}
		if = {
			limit = { DA_research_event_is_current = { id = 9 } }
			custom_tooltip = DA_Cost_10_ADK
			hidden_effect = {
				subtract_variable = {
					which = ancientDwarvenKnowledge
					value = 10
				}
			}
			DA_Research_add = { amount = 6 }
		}
		else = {
			custom_tooltip = DA_research_event_expired_tt
		}
	}
	option = {
		name = DA_Research_events.9.c
		highlight = yes
		trigger = {
			OR = {
				innovativeness = 95
				full_idea_group = innovativeness_ideas
			}
		}
	}
	after = {
		clr_country_flag = DA_research_event_pending.9
	}
}

	id = DA_Secret_Societies_event.3
	title = DA_SS_event_3_t
	desc = DA_SS_event_3_d
	picture = ADVISOR_eventPicture
	is_triggered_only = yes
	trigger = {
		NOT = {
			DA_SS_hammers_discovered = yes
		}

		has_country_flag = DA_SS_hammers_contact_pending
	}

	immediate = {
		random_owned_province = {
			limit = {
				DA_SS_forge_site = yes
			}

			save_event_target_as = DA_SS_contact_province
		}
	}

	option = {
		name = DA_SS_hammers_accept
		trigger = {
			event_target:DA_SS_contact_province = {
				owned_by = ROOT
				controlled_by = ROOT
			}
		}

		custom_tooltip = DA_SS_hammers_accept_details_tt
		hidden_effect = {
			event_target:DA_SS_contact_province = {
				DA_SS_establish_sanctuary_site = yes
			}

			set_country_flag = DA_SC_unlocked.Hammers_of_Halanna
			clr_country_flag = DA_SS_hammers_contact_pending
			DA_SS_hammers_become_1 = yes
			change_variable = {
				which = DA_Hammers_of_Halanna_relation
				value = 10
			}

			if = {
				limit = {
					check_variable = {
						which = DA_SS_hammers_cap
						value = 100
					}

					check_variable = {
						which = DA_Hammers_of_Halanna_relation
						which = DA_SS_hammers_cap
					}
				}

				set_variable = {
					which = DA_Hammers_of_Halanna_relation
					which = DA_SS_hammers_cap
				}
			}
		}
	}

	option = {
		name = DA_SS_hammers_decline
		ai_chance = {
			factor = 0
		}

		custom_tooltip = DA_SS_hammers_decline_details_tt

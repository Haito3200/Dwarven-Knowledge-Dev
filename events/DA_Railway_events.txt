namespace = DA_railway_event

# Railway upgrades: the reports of the workshop of the Railway Control Center (Er-Natvir).
# Rewritten on 11 November 2026. Each is queued by its button in the wonder window
# (common/custom_gui/DA_Wonder_Railway_HQ.txt) when the workshop is paid, and arrives when the drawings are done.
# .5 to .8 the first four upgrades (the hall at tier I), .9 to .12 the second four (tier III, and the first of its kind).
# The flag an event sets is what the building DA_railway_work reads (common/buildings/DA_buildings.txt).

country_event = {
	id = DA_railway_event.5 # Improved Railway I
	title = "DA_railway_event.5.t"
	desc = "DA_railway_event.5.d"
	picture = INVENTION_eventPicture

	is_triggered_only = yes
	immediate = {
		hidden_effect = {
			clr_country_flag = DA_RW_res_faster1
		}
	}
	option = {
		name = "DA_railway_event.research.a"
		custom_tooltip = DA_unlocked_rail_upgrade.tt
		tooltip = {
			random_core_province = {
				add_permanent_province_modifier = {
					name = DA_rail_faster_mod_1
					duration = -1
				}
			}
		}
		hidden_effect = {
			set_country_flag = DA_rail_faster1_c
			set_country_flag = DA_can_upgrade_rails
		}
	}
}

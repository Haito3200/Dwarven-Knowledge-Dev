country_event = { # the broken canals: a tremor
	id = DA_Magma_forge_event.4
	title = "DA_Magma_forge_event.4.t"
	desc = "DA_Magma_forge_event.4.d"
	picture = DA_Fireglass_Forge_broken_eventPicture

	is_triggered_only = yes
	trigger = {
		DA_FG_active = yes
		NOT = { DA_FG_repaired = yes }
		any_owned_province = { DA_FG_tremor_target = yes }
	}
	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = { DA_FG_tremor_target = yes }
				save_event_target_as = DA_target_province
			}
		}
	}

	option = {
		name = "DA_Magma_forge_event.4.a"
		goto = DA_target_province
		event_target:DA_target_province = {
			if = {
				limit = { devastation = 10 NOT = { province_id = 4039 } NOT = { has_province_modifier = DA_regulated_magma_vein } }
				add_permanent_province_modifier = { name = DA_unregulated_magma_vein duration = -1 }
			}
			add_devastation = 25
			hidden_effect = { add_province_modifier = { name = DA_MF_devastation_cooldown duration = 395 hidden = yes } }
		}
		custom_tooltip = DA_Magma_forge_event.4.tt
		hidden_effect = { DA_FG_count_veins = yes }
	}
	option = {
		name = "DA_Magma_forge_event.4.b"
		trigger = { treasury = 150 }
		goto = DA_target_province
		add_treasury = -150
		event_target:DA_target_province = {
			add_devastation = 10
			hidden_effect = { add_province_modifier = { name = DA_MF_devastation_cooldown duration = 395 hidden = yes } }
		}
	}
}

namespace = DA_Expedition_camp_event

# Secret passages: extra events of Anbennar's dungeon expeditions, for players only.
# common/on_actions/DA_on_actions.txt (on_yearly_pulse_4) fires .3 once a year in one of the player's provinces with an
# expedition in progress. Exploring rolls one outcome with DA_SP_roll_passage (common/scripted_effects/DA_Secret_Passage_effects.txt),
# where the weights are. The party effects (change_party_*) are Anbennar's.
# Description: Claude files/21 - Secret passages.md. Tester: DA_Secret_Passage_tester.txt.

############################################################
# The passage
############################################################

province_event = {
	id = DA_Expedition_camp_event.3			# secret passage: explore, scout first, or ignore
	title = DA_Expedition_camp_event.3.t
	desc = DA_Expedition_camp_event.3.d
	picture = CAVE_eventPicture
	is_triggered_only = yes

	option = {
		name = DA_Expedition_camp_event.3.a			# explore
		custom_tooltip = DA_Expedition_camp_event.3.tooltip
		if = {
			limit = { has_province_flag = DA_passage_forewarned }
			custom_tooltip = DA_SP_forewarned_tt
		}
		change_party_supplies = { remove_tooltip = yes loss = 6 }
		hidden_effect = {
			if = {
				limit = { has_province_flag = DA_passage_forewarned }
				set_province_flag = DA_passage_scouted
				clr_province_flag = DA_passage_forewarned
			}
			DA_SP_roll_passage = yes
			clr_province_flag = DA_passage_scouted
		}
	}
	option = {
		name = DA_Expedition_camp_event.3.b			# send scouts ahead
		trigger = {
			NOT = { has_province_flag = DA_passage_forewarned }
		}
		custom_tooltip = DA_SP_scouts_tt
		change_party_supplies = { remove_tooltip = yes loss = 8 }
		hidden_effect = {
			set_province_flag = DA_passage_scouted
			DA_SP_roll_passage = yes
			clr_province_flag = DA_passage_scouted
		}
	}
	option = {
		name = DA_Expedition_camp_event.3.coward		# ignore it
	}
}
